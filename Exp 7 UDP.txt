server.py
import socket

# Create a UDP socket
server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Bind the server to an IP address and port
server.bind(("127.0.0.1", 12345))

print("UDP Server is waiting for a file...")

# Receive the file name
filename, client_address = server.recvfrom(1024)
filename = filename.decode()

print("Receiving file:", filename)

# Open a new file to save the received data
with open("received_" + filename, "wb") as file:
    while True:
        # Receive file data
        data, client_address = server.recvfrom(1024)

        # Check for end-of-file marker
        if data == b"EOF":
            break

        # Write received data to the file
        file.write(data)

print("File received successfully!")
server.close()

Client.py
import socket
import os

# Create a UDP socket
client = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Server address and port
server_address = ("127.0.0.1", 12345)

# Name of the file to send
filename = "sample.txt"

# Send the file name to the server
client.sendto(filename.encode(), server_address)

# Open the file in binary mode
with open(filename, "rb") as file:
    while True:
        # Read file data in blocks
        data = file.read(1024)

        # Stop when the file ends
        if not data:
            break

        # Send the data to the server
        client.sendto(data, server_address)

# Send end-of-file marker
client.sendto(b"EOF", server_address)

print("File sent successfully!")
client.close()
